% --- Core theorem environment style ---
\tcbset{
    my_thm_base/.style={
        enhanced,
        breakable,
        arc=6pt,
        outer arc=6pt,
        boxrule=1pt,
        width=\textwidth,
        center,
        left=10pt, right=10pt, top=15pt, bottom=8pt,
        before skip=22pt, after skip=20pt,
        fontupper=\rmfamily,          % Body text uses serif font (Song)
        fonttitle=\bfseries\sffamily, % Titles use sans-serif + bold (DengXian)
        coltitle=white,
        separator sign none,
        titlerule=0pt,
        toptitle=1mm, bottomtitle=1mm,
    }
}

% Definition environment (green)
\newtcbtheorem[number within=section]{definition}{定义}{%
    my_thm_base, colframe=defgreen, colback=defgreen!5,
    separator sign={：},
    title={\thesection.\arabic{tcbcounter}\quad\tcbtitle}
}{def}

% Notation environment (green)
\newtcbtheorem[number within=section, use counter from=definition]{notation}{记号}{%
    my_thm_base, colframe=defgreen, colback=defgreen!5,
    separator sign={：},
    title={\thesection.\arabic{tcbcounter}\quad\tcbtitle}
}{not}

% Theorem environment (orange)
\newtcbtheorem[number within=section]{theorem}{定理}{%
    my_thm_base, colframe=thmorange, colback=thmorange!5,
    separator sign={：},
    title={\thesection.\arabic{tcbcounter}\quad\tcbtitle}
}{thm}

% Lemma environment (orange)
\newtcbtheorem[number within=section, use counter from=theorem]{lemma}{引理}{%
    my_thm_base, colframe=thmorange, colback=thmorange!5,
    separator sign={：},
    title={\thesection.\arabic{tcbcounter}\quad\tcbtitle}
}{lem}

% Proposition environment (orange)
\newtcbtheorem[number within=section, use counter from=theorem]{proposition}{命题}{%
    my_thm_base, colframe=thmorange, colback=thmorange!5,
    separator sign={：},
    title={\thesection.\arabic{tcbcounter}\quad\tcbtitle}
}{prop}

% Example environment (horizontal lines style)
\newtcbtheorem[number within=section]{example}{}{%
    enhanced,
    breakable,
    frame hidden,             % Hide outer frame
    interior hidden,          % Hide background color
    sharp corners,            % Sharp corners
    boxrule=0pt,              % Base border thickness
    left=15pt, right=15pt,    % Left/right padding
    top=12pt, bottom=12pt,    % Internal vertical spacing
    toptitle=2pt,            % Spacing between title and top line
    before skip=12pt,        % Spacing before box
    after skip=12pt,         % Spacing after box
    % Top and bottom horizontal lines (gray start, black end)
    borderline north={1pt}{0pt}{black!50},
    borderline south={1pt}{0pt}{black},
    coltitle=black,               % Title color
    fonttitle=\bfseries\sffamily, % Sans-serif bold title
    fontupper=\rmfamily,          % Body text uses serif
    attach title to upper,
    after title={\par\smallskip},
    separator sign none,
    title={\thesection.\arabic{tcbcounter}\quad 例：\tcbtitle},
}{ex}

% Exercise environment (brown)
\newtcbtheorem[number within=section]{exercise}{练习}{%
    my_thm_base, colframe=exercisehead, colback=exercisehead!5,
    separator sign={：},
    title={\thesection.\arabic{tcbcounter}\quad\tcbtitle}
}{exer}

% Remark environment (dark gray)
\newtcbtheorem[number within=section]{remark}{注}{%
    my_thm_base, colframe=remgray, colback=remgray!5,
    separator sign={：},
    title={\thesection.\arabic{tcbcounter}\quad\tcbtitle}
}{rem}

}
